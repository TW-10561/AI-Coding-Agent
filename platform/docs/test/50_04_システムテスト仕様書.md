### 3.5 パフォーマンス計測方法

```typescript
// Latency measurement
async function measureLatency(fn: () => Promise<any>, iterations = 100) {
  const times: number[] = []
  for (let i = 0; i < iterations; i++) {
    const start = process.hrtime.bigint()
    await fn()
    const end = process.hrtime.bigint()
    times.push(Number(end - start) / 1_000_000) // ms
  }
  times.sort((a, b) => a - b)
  return {
    min: times[0],
    p50: times[Math.floor(iterations * 0.5)],
    p95: times[Math.floor(iterations * 0.95)],
    p99: times[Math.floor(iterations * 0.99)],
    max: times[times.length - 1],
    avg: times.reduce((a, b) => a + b, 0) / iterations,
  }
}
```

## 6. 回帰テストスイート

### 6.1 CI/CD 統合

```yaml
# .github/workflows/test.yml
name: Test Suite
on: [push, pull_request]
jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: oven-sh/setup-bun@v2
      - run: bun install
      - run: bun test --filter "unit"
      - run: bun test --filter "integration"
      - run: bun test --coverage
```

## 7. テスト環境構築手順

```bash
# 1. テスト環境セットアップ
cd platform
bun install

# 2. テスト用データベース初期化
export SQLITE_PATH=":memory:"

# 3. Mock vLLM 起動
bun run tests/mocks/mock-vllm.ts &

# 4. Platform テストモード起動
NODE_ENV=test bun run src/server/index.ts &

# 5. テスト実行
bun test                          # All tests
bun test --filter "unit"          # Unit only
bun test --filter "integration"   # Integration only
bun test --filter "security"      # Security only
bun test --coverage               # With coverage report

# 6. パフォーマンステスト
bun run tests/perf/benchmark.ts
```
